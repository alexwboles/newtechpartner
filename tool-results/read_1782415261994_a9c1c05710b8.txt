    59→/* ─── ANIMATED SECTION WRAPPER ─── */
    60→function AnimatedSection({
    61→  children,
    62→  className = "",
    63→  delay = 0,
    64→}: {
    65→  children: ReactNode;
    66→  className?: string;
    67→  delay?: number;
    68→}) {
    69→  const ref = useRef(null);
    70→  const isInView = useInView(ref, { once: true, margin: "-80px" });
    71→  return (
    72→    <motion.div
    73→      ref={ref}
    74→      initial={{ opacity: 0, y: 40 }}
    75→      animate={isInView ? { opacity: 1, y: 0 } : {}}
    76→      transition={{ duration: 0.7, delay, ease: "easeOut" }}
    77→      className={className}
    78→    >
    79→      {children}
    80→    </motion.div>
    81→  );
    82→}

   117→/* ─── COUNT UP ANIMATION ─── */
   118→function CountUp({ target, suffix = "", prefix = "" }: { target: number; suffix?: string; prefix?: string }) {
   119→  const [count, setCount] = useState(0);
   120→  const ref = useRef(null);
   121→  const isInView = useInView(ref, { once: true });
   122→
   123→  useEffect(() => {
   124→    if (!isInView) return;
   125→    const duration = 2000;
   126→    const steps = 60;
   127→    const increment = target / steps;
   128→    let current = 0;
   129→    const timer = setInterval(() => {
   130→      current += increment;
   131→      if (current >= target) {
   132→        setCount(target);
   133→        clearInterval(timer);
   134→      } else {
   135→        setCount(Math.floor(current));
   136→      }
   137→    }, duration / steps);
   138→    return () => clearInterval(timer);
   139→  }, [isInView, target]);
   140→
   141→  return (
   142→    <span ref={ref}>
   143→      {prefix}
   144→      {count.toLocaleString()}
   145→      {suffix}
   146→    </span>
   147→  );
   148→}
